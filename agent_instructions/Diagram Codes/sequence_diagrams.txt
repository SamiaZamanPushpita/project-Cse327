5. User Authentication
@startuml
title User Authentication Sequence Diagram


actor User
boundary "Login Page" as UI
control "Authentication Controller" as Auth
database "User Database" as DB


User -> UI : Enter email & password
activate UI


UI -> Auth : authenticate(credentials)
activate Auth


Auth -> DB : validateUser(credentials)
activate DB


alt Valid Credentials
    DB --> Auth : User Record
    deactivate DB


    Auth --> UI : Authentication Successful
    deactivate Auth


    UI --> User : Redirect to Dashboard


else Invalid Credentials
    DB --> Auth : User Not Found
    deactivate DB


    Auth --> UI : Authentication Failed
    deactivate Auth


    UI --> User : Display Error Message
end


deactivate UI


@enduml
