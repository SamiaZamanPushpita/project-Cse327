1. Authentication & User Management Use Case
@startuml
left to right direction


title Authentication & User Management Use Case Diagram


skinparam monochrome true
skinparam shadowing false
skinparam packageStyle rectangle


actor Guest
actor User
actor Tutor
actor Student
actor Parent


User <|-- Tutor
User <|-- Student
User <|-- Parent


rectangle "Tutor Management System" {


    usecase "Register" as UC1
    usecase "Login" as UC2
    usecase "Login with Google" as UC3
    usecase "Forgot Password" as UC4


    usecase "Logout" as UC5
    usecase "Update Profile" as UC6
    usecase "Change Password" as UC7
}


Guest --> UC1
Guest --> UC2
Guest --> UC3
Guest --> UC4


User --> UC5
User --> UC6
User --> UC7


@enduml
